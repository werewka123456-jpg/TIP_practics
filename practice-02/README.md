export function findTaskById(tasks, id) {
  return tasks.find((task) => task.id === id);
}

export function getPendingTasks(tasks) {
  return tasks.filter((task) => task.completed === false);
}

export function getTaskTitles(tasks) {
  return tasks.map((task) => task.title);
}

export function getTaskStats(tasks) {
  const total = tasks.length;
  const completed = tasks.filter((task) => task.completed === true).length;
  const pending = total - completed;
  const progress = total > 0 ? (completed / total) * 100 : 0;
  return { total, completed, pending, progress };
}

Почему id не равен индексу:
id - это идентификатор записи, который не меняется при удалении соседних элементов. Индекс - это текущее место в массиве, и он меняется при добавлении/удалении. Поиск задачи по id нельзя заменять обращением tasks[id] - это даст неверный результат. Именно поэтому используется find, а не доступ по индексу.

Почему getTaskStats возвращает число, а не строку с процентом:
Функция - расчётная, она возвращает данные для дальнейшей обработки. Округление и форматирование (toFixed(1), %) - задача отображения в main.js. Если бы getTaskStats возвращала строку, её нельзя было бы использовать в других расчётах или сравнениях.

Граница обязательного контракта:
Аргумент tasks во всех функциях этой работы - корректный обычный массив задач без пропусков и с уникальными id. Проверять заново каждое поле всех существующих записей при каждом чтении не требуется. Корректность новых данных обеспечивается функциями создания и изменения.

Строковый id:
Для поиска строка "4" не совпадает с числом 4: результатом будет undefined. Это правильно, потому что findTaskById использует строгое сравнение ===, без приведения типов.

### Наиболее показательный момент

Наиболее показательным оказался эксперимент 2 из Задания 1 — стрелочная функция с телом в `{}` без `return` возвращает `undefined`. Это демонстрирует важное отличие стрелочной функции с кратким телом (`value => value * value`) от функции с телом в фигурных скобках (`value => { return value * value; }`). Ошибка незаметна без запуска, потому что код «выглядит правильным», но не возвращает результат.

### Что изменилось по сравнению с ПР1

В ПР1 расчёты выполнялись над отдельными переменными. В ПР2 данные организованы в массив объектов, а операции — в функции. Это позволило:
- избежать дублирования кода;
- обеспечить неизменность исходных данных;
- разделить ответственность между файлами (данные — логика — сценарий — проверки);
- подготовить основу для работы с DOM в ПР3.
